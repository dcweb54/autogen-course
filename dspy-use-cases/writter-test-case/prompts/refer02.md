def mix_audio(voice_paths, music_path, output_path):
    music = AudioSegment.from_file(music_path)
    final = AudioSegment.silent(duration=0)
    
    for vp in voice_paths:
        voice = AudioSegment.from_file(vp['path'])
        # Duck music under voice

        mixed = voice.overlay(music_section)
        final += mixed
